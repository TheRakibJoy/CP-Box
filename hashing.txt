vl H(N),p(N);
void compute_hash(string &s)
{
    p[0]=1;
    ll base=61;
    for(int i=1; i<N; i++)
    {
        p[i]=(p[i-1]*base)%mod;
    }
    ll sz=s.size();
    ll val=0;
    for(int i=1; i<=sz; i++)
    {
        val*=base;
        val+=(s[i-1]-'a'+1);
        val%=mod;
        H[i]=val;
    }
}
ll hash_value(ll i,ll j)
{
    return (H[j]-(H[i-1]*p[j-i+1])%mod+mod)%mod;
}
