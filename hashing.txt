vi H(N+5),p(N+5);
void compute_hash(string &s)
{
    p[0]=1;
    ll base=61;
    for(int i=1; i<=N; i++)
    {
        p[i]=(p[i-1]*base)%mod;
    }
    ll sz=s.size();
    ll val=0;
    for(int i=1; i<=sz; i++)
    {
        val*=base;
        val+=(s[i-1]-'a'+1);
        val%=mod;
        H[i]=val;
    }
}
int hash_value(ll i,ll j)
{
    ll an=1LL*H[j]-(1LL*H[i-1]*p[j-i+1])%mod;
    if(an<0) an+=mod;
    return an;
}
