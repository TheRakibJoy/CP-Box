bool prime[N+5];
void primenire()
{
    int i, j;
    prime[0]=true;
    prime[1]=true;
    for(i=4; i<=1000000; i+=2)
    {
        prime[i]=true;
    }
    for(i=3; i<=1001; i+=2)
    {
        if(!prime[i])
        {
            for(j=i*i; j<=1000001; j+=2*i)
            {
                prime[j]=true;
            }
        }
    }
}
